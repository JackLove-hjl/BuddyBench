 172 | <script>
 173 | (function () {
 174 |   "use strict";
 175 | 
 176 |   // ===== 配置 =====
 177 |   const GRID = 20;          // 20x20 网格
 178 |   const CELL = 20;          // 每格 20px，画布 400x400
 179 |   const STEP_MS = 120;      // 每步间隔(毫秒)
 180 | 
 181 |   const canvas = document.getElementById('game');
 182 |   const ctx = canvas.getContext('2d');
 183 |   const scoreEl = document.getElementById('score');
 184 |   const finalScoreEl = document.getElementById('final-score');
 185 |   const ovStart = document.getElementById('ov-start');
 186 |   const ovOver = document.getElementById('ov-over');
 187 |   const dpad = document.getElementById('dpad');
 188 | 
 189 |   // ---- 状态 ----
 190 |   let snake, dir, nextDir, food, score;
 191 |   let state = 'start';      // start | playing | paused | over
 192 |   let lastTick = 0;
 193 |   let rafId = null;
 194 |   let dpr = 1;
 195 | 
 196 |   const DIR_VEC = {
 197 |     up:    { x: 0, y: -1 },
 198 |     down:  { x: 0, y: 1 },
 199 |     left:  { x: -1, y: 0 },
 200 |     right: { x: 1, y: 0 }
 201 |   };
 202 | 
 203 |   // ---- 画布(适配高分屏) ----
 204 |   function setupCanvas() {
 205 |     dpr = window.devicePixelRatio || 1;
 206 |     canvas.width = GRID * CELL * dpr;
 207 |     canvas.height = GRID * CELL * dpr;
 208 |     canvas.style.width = (GRID * CELL) + 'px';
 209 |     canvas.style.height = (GRID * CELL) + 'px';
 210 |   }
 211 | 
 212 |   // ---- 新游戏 ----
 213 |   function resetGame() {
 214 |     snake = [{ x: 10, y: 10 }, { x: 9, y: 10 }, { x: 8, y: 10 }];
 215 |     dir = { x: 1, y: 0 };
 216 |     nextDir = { x: 1, y: 0 };
 217 |     score = 0;
 218 |     scoreEl.textContent = '0';
 219 |     spawnFood();
 220 |     draw();
 221 |   }
 222 | 
 223 |   function spawnFood() {
 224 |     const occupied = new Set(snake.map(s => s.x + ',' + s.y));
 225 |     const free = [];
 226 |     for (let x = 0; x < GRID; x++)
 227 |       for (let y = 0; y < GRID; y++)
 228 |         if (!occupied.has(x + ',' + y)) free.push({ x, y });
 229 |     if (free.length === 0) { food = null; return; }
 230 |     food = free[Math.floor(Math.random() * free.length)];
 231 |   }
 232 | 
 233 |   // ---- 移动一步 ----
 234 |   function step() {
 235 |     if (state !== 'playing') return;
 236 | 
 237 |     // 应用缓冲方向，禁止 180° 掉头
 238 |     const opp = { x: -dir.x, y: -dir.y };
 239 |     if (!(nextDir.x === opp.x && nextDir.y === opp.y)) {
 240 |       dir = { x: nextDir.x, y: nextDir.y };
 241 |     }
 242 | 
 243 |     const head = snake[0];
 244 |     const newHead = { x: head.x + dir.x, y: head.y + dir.y };
 245 | 
 246 |     // 撞墙
 247 |     if (newHead.x < 0 || newHead.x >= GRID || newHead.y < 0 || newHead.y >= GRID) {
 248 |       gameOver();
 249 |       return;
 250 |     }
 251 | 
 252 |     // 撞自身(尾部即将移开时不算)
 253 |     const willEat = food && newHead.x === food.x && newHead.y === food.y;
 254 |     const body = willEat ? snake : snake.slice(0, -1);
 255 |     if (body.some(s => s.x === newHead.x && s.y === newHead.y)) {
 256 |       gameOver();
 257 |       return;
 258 |     }
 259 | 
 260 |     snake.unshift(newHead);
 261 |     if (willEat) {
 262 |       score++;
 263 |       scoreEl.textContent = score;
 264 |       spawnFood();
 265 |     } else {
 266 |       snake.pop();
 267 |     }
 268 | 
 269 |     draw();
 270 |   }
 271 | 
 272 |   // ---- 主循环 ----
 273 |   function loop(ts) {
 274 |     if (state === 'playing' && (ts - lastTick) >= STEP_MS) {
 275 |       step();
 276 |       lastTick = ts;
 277 |     }
 278 |     rafId = requestAnimationFrame(loop);
 279 |   }
 280 | 
 281 |   // ---- 状态切换 ----
 282 |   function startGame() {
 283 |     resetGame();
 284 |     state = 'playing';
 285 |     ovStart.classList.add('hidden');
 286 |     ovOver.classList.add('hidden');
 287 |     if (!rafId) {
 288 |       lastTick = 0;
 289 |       rafId = requestAnimationFrame(loop);
 290 |     }
 291 |   }
 292 | 
 293 |   function togglePause() {
 294 |     if (state === 'playing') {
 295 |       state = 'paused';
 296 |     } else if (state === 'paused') {
 297 |       state = 'playing';
 298 |       lastTick = 0;   // 避免恢复后立即走一步
 299 |     }
 300 |   }
 301 | 
 302 |   function gameOver() {
 303 |     state = 'over';
 304 |     finalScoreEl.textContent = score;
 305 |     ovOver.classList.remove('hidden');
 306 |   }
 307 | 
 308 |   // ---- 绘制 ----
 309 |   function draw() {
 310 |     ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
 311 |     const size = GRID * CELL;
 312 | 
 313 |     // 背景
 314 |     ctx.fillStyle = '#0b1220';
 315 |     ctx.fillRect(0, 0, size, size);
 316 | 
 317 |     // 食物
 318 |     if (food) {
 319 |       const fx = food.x * CELL + CELL / 2;
 320 |       const fy = food.y * CELL + CELL / 2;
 321 |       ctx.fillStyle = '#ef4444';
 322 |       ctx.beginPath();
 323 |       ctx.arc(fx, fy, CELL / 2 - 2, 0, Math.PI * 2);
 324 |       ctx.fill();
 325 |     }
 326 | 
 327 |     // 蛇身(头→尾 青→绿渐变)
 328 |     for (let i = snake.length - 1; i >= 0; i--) {
 329 |       const s = snake[i];
 330 |       const t = i / Math.max(snake.length - 1, 1);
 331 |       const pad = i === 0 ? 1 : 3;
 332 |       const x = s.x * CELL + pad;
 333 |       const y = s.y * CELL + pad;
 334 |       const w = CELL - pad * 2;
 335 |       ctx.fillStyle = i === 0 ? '#67e8f9' : `hsl(${160 - t * 40}, 70%, ${55 - t * 10}%)`;
 336 |       roundRect(ctx, x, y, w, w, i === 0 ? 6 : 5);
 337 |       ctx.fill();
 338 |     }
 339 | 
 340 |     // 蛇眼睛
 341 |     const head = snake[0];
 342 |     if (head) {
 343 |       const cx = head.x * CELL + CELL / 2;
 344 |       const cy = head.y * CELL + CELL / 2;
 345 |       ctx.fillStyle = '#0b1220';
 346 |       const off = 4, fwd = 3;
 347 |       let ex1, ey1, ex2, ey2;
 348 |       if (dir.x !== 0) { ex1 = cx + dir.x * fwd; ey1 = cy - off; ex2 = cx + dir.x * fwd; ey2 = cy + off; }
 349 |       else              { ex1 = cx - off; ey1 = cy + dir.y * fwd; ex2 = cx + off; ey2 = cy + dir.y * fwd; }
 350 |       ctx.beginPath(); ctx.arc(ex1, ey1, 2, 0, Math.PI * 2); ctx.fill();
 351 |       ctx.beginPath(); ctx.arc(ex2, ey2, 2, 0, Math.PI * 2); ctx.fill();
 352 |     }
 353 |   }
 354 | 
 355 |   function roundRect(ctx, x, y, w, h, r) {
 356 |     r = Math.min(r, w / 2, h / 2);
 357 |     ctx.beginPath();
 358 |     ctx.moveTo(x + r, y);
 359 |     ctx.arcTo(x + w, y, x + w, y + h, r);
 360 |     ctx.arcTo(x + w, y + h, x, y + h, r);
 361 |     ctx.arcTo(x, y + h, x, y, r);
 362 |     ctx.arcTo(x, y, x + w, y, r);
 363 |     ctx.closePath();
 364 |   }
 365 | 
 366 |   // ---- 输入 ----
 367 |   function setDirection(name) {
 368 |     const v = DIR_VEC[name];
 369 |     if (v) nextDir = v;
 370 |   }
 371 | 
 372 |   document.addEventListener('keydown', function (e) {
 373 |     const key = e.key;
 374 |     if (key === 'ArrowUp' || key === 'w' || key === 'W') { e.preventDefault(); if (state !== 'start' && state !== 'over') setDirection('up'); return; }
 375 |     if (key === 'ArrowDown' || key === 's' || key === 'S') { e.preventDefault(); if (state !== 'start' && state !== 'over') setDirection('down'); return; }
 376 |     if (key === 'ArrowLeft' || key === 'a' || key === 'A') { e.preventDefault(); if (state !== 'start' && state !== 'over') setDirection('left'); return; }
 377 |     if (key === 'ArrowRight' || key === 'd' || key === 'D') { e.preventDefault(); if (state !== 'start' && state !== 'over') setDirection('right'); return; }
 378 | 
 379 |     if (key === ' ') { e.preventDefault(); if (state === 'playing' || state === 'paused') togglePause(); return; }
 380 | 
 381 |     if (key === 'Enter') {
 382 |       e.preventDefault();
 383 |       if (state === 'start' || state === 'over') startGame();
 384 |     }
 385 |   });
 386 | 
 387 |   dpad.addEventListener('click', function (e) {
 388 |     const btn = e.target.closest('button[data-dir]');
 389 |     if (!btn) return;
 390 |     setDirection(btn.dataset.dir);
 391 |   });
 392 | 
 393 |   document.getElementById('btn-start').addEventListener('click', startGame);
 394 |   document.getElementById('btn-again').addEventListener('click', startGame);
 395 | 
 396 |   // 失焦自动暂停
 397 |   document.addEventListener('visibilitychange', function () {
 398 |     if (document.hidden && state === 'playing') togglePause();
 399 |   });
 400 | 
 401 |   // ---- 启动 ----
 402 |   setupCanvas();
 403 |   resetGame();
 404 | })();
 405 | </script>
