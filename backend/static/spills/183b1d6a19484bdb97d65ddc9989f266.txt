 373 |   // 生成食物(随机空格点)
 374 |   function spawnFood() {
 375 |     const occupied = new Set(snake.map(s => s.x + ',' + s.y));
 376 |     const free = [];
 377 |     for (let x = 0; x < GRID; x++) {
 378 |       for (let y = 0; y < GRID; y++) {
 379 |         if (!occupied.has(x + ',' + y)) free.push({ x, y });
 380 |       }
 381 |     }
 382 |     if (free.length === 0) { food = null; return; }  // 棋盘已满 → 胜利
 383 |     food = free[Math.floor(Math.random() * free.length)];
 384 |   }

 386 |   // ============ 游戏主循环(固定时间步长) ============
 387 |   function step() {
 388 |     if (state !== 'playing') return;
 389 | 
 390 |     // 应用缓冲方向(禁止 180° 掉头)
 391 |     const opp = { x: -dir.x, y: -dir.y };
 392 |     if (!(nextDir.x === opp.x && nextDir.y === opp.y)) {
 393 |       dir = { x: nextDir.x, y: nextDir.y };
 394 |     }
 395 | 
 396 |     const head = snake[0];
 397 |     const newHead = { x: head.x + dir.x, y: head.y + dir.y };
 398 | 
 399 |     // 撞墙判定
 400 |     if (newHead.x < 0 || newHead.x >= GRID || newHead.y < 0 || newHead.y >= GRID) {
 401 |       gameOver();
 402 |       return;
 403 |     }
 404 | 
 405 |     // 撞自身判定(注意:若新头恰好在尾部且尾部即将移动则不判死，但这里简化处理)
 406 |     const willEat = food && newHead.x === food.x && newHead.y === food.y;
 407 |     const bodyToCheck = willEat ? snake : snake.slice(0, -1);
 408 |     if (bodyToCheck.some(s => s.x === newHead.x && s.y === newHead.y)) {
 409 |       gameOver();
 410 |       return;
 411 |     }
 412 | 
 413 |     // 移动
 414 |     snake.unshift(newHead);
 415 |     if (willEat) {
 416 |       score++;
 417 |       scoreEl.textContent = score;
 418 |       spawnFood();
 419 |       if (!food) { win(); return; }
 420 |     } else {
 421 |       snake.pop();
 422 |     }
 423 | 
 424 |     draw();
 425 |   }

 479 |   // ============ 绘制 ============
 480 |   function draw() {
 481 |     ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
 482 |     const size = GRID * CELL;
 483 | 
 484 |     // 背景
 485 |     ctx.fillStyle = '#0b1220';
 486 |     ctx.fillRect(0, 0, size, size);
 487 | 
 488 |     // 网格线(淡)
 489 |     ctx.strokeStyle = 'rgba(255,255,255,0.04)';
 490 |     ctx.lineWidth = 1;
 491 |     for (let i = 1; i < GRID; i++) {
 492 |       ctx.beginPath();
 493 |       ctx.moveTo(i * CELL, 0);
 494 |       ctx.lineTo(i * CELL, size);
 495 |       ctx.stroke();
 496 |       ctx.beginPath();
 497 |       ctx.moveTo(0, i * CELL);
 498 |       ctx.lineTo(size, i * CELL);
 499 |       ctx.stroke();
 500 |     }
 501 | 
 502 |     // 食物(带光晕的渐变圆)
 503 |     if (food) {
 504 |       const fx = food.x * CELL + CELL / 2;
 505 |       const fy = food.y * CELL + CELL / 2;
 506 |       ctx.shadowColor = 'rgba(248, 113, 113, 0.9)';
 507 |       ctx.shadowBlur = 12;
 508 |       const grad = ctx.createRadialGradient(fx - 3, fy - 3, 2, fx, fy, CELL / 2 + 1);
 509 |       grad.addColorStop(0, '#fca5a5');
 510 |       grad.addColorStop(1, '#ef4444');
 511 |       ctx.fillStyle = grad;
 512 |       ctx.beginPath();
 513 |       ctx.arc(fx, fy, CELL / 2 - 2, 0, Math.PI * 2);
 514 |       ctx.fill();
 515 |       ctx.shadowBlur = 0;
 516 |     }
 517 | 
 518 |     // 蛇身
 519 |     for (let i = snake.length - 1; i >= 0; i--) {
 520 |       const s = snake[i];
 521 |       const t = i / Math.max(snake.length - 1, 1);   // 0=头 1=尾
 522 |       const pad = i === 0 ? 1 : 3;                    // 头部略大
 523 |       const x = s.x * CELL + pad;
 524 |       const y = s.y * CELL + pad;
 525 |       const w = CELL - pad * 2;
 526 | 
 527 |       // 头→尾 颜色渐变(青 → 绿)
 528 |       const isHead = i === 0;
 529 |       if (isHead) {
 530 |         ctx.shadowColor = 'rgba(34, 211, 238, 0.8)';
 531 |         ctx.shadowBlur = 10;
 532 |       } else {
 533 |         ctx.shadowBlur = 0;
 534 |       }
 535 | 
 536 |       ctx.fillStyle = isHead
 537 |         ? '#67e8f9'
 538 |         : `hsl(${160 - t * 40}, 70%, ${55 - t * 10}%)`;
 539 | 
 540 |       // 圆角矩形
 541 |       roundRect(ctx, x, y, w, w, isHead ? 6 : 5);
 542 |       ctx.fill();
 543 |     }
 544 |     ctx.shadowBlur = 0;
 545 | 
 546 |     // 蛇眼睛(仅头部)
 547 |     const head = snake[0];
 548 |     if (head) {
 549 |       const cx = head.x * CELL + CELL / 2;
 550 |       const cy = head.y * CELL + CELL / 2;
 551 |       ctx.fillStyle = '#0b1220';
 552 |       const eyeOff = 4;
 553 |       const fwd = 3;
 554 |       // 眼睛位置随方向
 555 |       let ex1, ey1, ex2, ey2;
 556 |       if (dir.x !== 0) { // 水平
 557 |         ex1 = cx + dir.x * fwd; ey1 = cy - eyeOff;
 558 |         ex2 = cx + dir.x * fwd; ey2 = cy + eyeOff;
 559 |       } else {           // 垂直
 560 |         ex1 = cx - eyeOff; ey1 = cy + dir.y * fwd;
 561 |         ex2 = cx + eyeOff; ey2 = cy + dir.y * fwd;
 562 |       }
 563 |       ctx.beginPath(); ctx.arc(ex1, ey1, 2, 0, Math.PI * 2); ctx.fill();
 564 |       ctx.beginPath(); ctx.arc(ex2, ey2, 2, 0, Math.PI * 2); ctx.fill();
 565 |     }
 566 |   }

 568 |   // 圆角矩形辅助
 569 |   function roundRect(ctx, x, y, w, h, r) {
 570 |     r = Math.min(r, w / 2, h / 2);
 571 |     ctx.beginPath();
 572 |     ctx.moveTo(x + r, y);
 573 |     ctx.arcTo(x + w, y, x + w, y + h, r);
 574 |     ctx.arcTo(x + w, y + h, x, y + h, r);
 575 |     ctx.arcTo(x, y + h, x, y, r);
 576 |     ctx.arcTo(x, y, x + w, y, r);
 577 |     ctx.closePath();
 578 |   }
